// ── Helpers ──────────────────────────────────────────────────────────────────
function timestampFromDate(dateString?: string | null): number {
  if (!dateString) return 0;
  const timestamp = new Date(dateString).getTime();
  return Number.isFinite(timestamp) ? timestamp : 0;
}

function formatTimeAgo(dateString?: string | null): string {
  const timestamp = timestampFromDate(dateString);
  if (!timestamp) return "Just now";

  const seconds = Math.floor((new Date().getTime() - timestamp) / 1000);
  
  if (seconds < 60) return "Just now";
  const minutes = Math.floor(seconds / 60);
  if (minutes < 60) return `${minutes} min${minutes > 1 ? "s" : ""} ago`;
  const hours = Math.floor(minutes / 60);
  if (hours < 24) return `${hours} hour${hours > 1 ? "s" : ""} ago`;
  const days = Math.floor(hours / 24);
  return `${days} day${days > 1 ? "s" : ""} ago`;
}

// ── Circular Progress Component ────────────────────────────────────────────────
function ProgressRing({ radius = 16, stroke = 3, progress = 0, color = "#1b59f8" }) {
  const normalizedRadius = radius - stroke * 2;
  const circumference = normalizedRadius * 2 * Math.PI;
  const strokeDashoffset = circumference - (Math.min(Math.max(progress, 0), 100) / 100) * circumference;

  return (
    <svg height={radius * 2} width={radius * 2} className="transform -rotate-90">
      <circle
        stroke="#f1f5f9"
        fill="transparent"
        strokeWidth={stroke}
        r={normalizedRadius}
        cx={radius}
        cy={radius}
      />
      <circle
        stroke={color}
        fill="transparent"
        strokeWidth={stroke}
        strokeDasharray={circumference + " " + circumference}
        style={{ strokeDashoffset }}
        strokeLinecap="round"
        r={normalizedRadius}
        cx={radius}
        cy={radius}
      />
    </svg>
  );
}

                <p className="text-lg font-bold text-slate-800 mt-1">{jiraStats.syncedCount}</p>
              </div>
              <div className="border-x border-slate-100">
                <p className="text-[10px] font-bold uppercase tracking-wider text-slate-400">
                  Sync Failures
                </p>
                <p className={cn("text-lg font-bold mt-1", jiraStats.failureCount > 0 ? "text-red-600" : "text-slate-800")}>
                  {jiraStats.failureCount}
                </p>
              </div>
              <div>
                <p className="text-[10px] font-bold uppercase tracking-wider text-slate-400">
                  Conflicts
                </p>
                <p className={cn("text-lg font-bold mt-1", jiraStats.conflictCount > 0 ? "text-amber-500" : "text-slate-800")}>
                  {jiraStats.conflictCount}
                </p>
              </div>
            </div>

            {/* Link to Sync Page */}
            <div className="mt-6">
              <Link href="/settings" className="text-xs font-semibold text-[#1b59f8] hover:underline flex items-center gap-1">
                View Sync Monitor
                <ArrowUpRight className="h-3.5 w-3.5" />
              </Link>
            </div>
          </CardContent>
        </Card>
      </div>

      {/* ── Row 2: AI Agent Activity, Pending Approvals, Recent Activity ───────── */}
      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        {/* 1. AI Agent Activity */}
        <Card className="border-slate-200">
          <CardHeader className="p-5 flex flex-row items-center justify-between border-b border-slate-100">
            <div>
              <CardTitle className="text-sm font-semibold">AI Agent Activity</CardTitle>
              <CardDescription className="text-[10px]">Real-time autonomous engineering agents</CardDescription>
            </div>
            <select className="border border-slate-200 rounded-lg text-[10px] font-semibold px-2 py-1 text-slate-500 focus:outline-none focus:ring-1 focus:ring-[#1b59f8]">
              <option>Recent Runs</option>
            </select>
          </CardHeader>
          <CardContent className="p-4 pt-2">
            <div className="space-y-4">
              {agentActivity.map((agent, i) => (
                <div key={`${agent.name}-${i}`} className="flex items-start gap-3.5 text-xs py-1.5">
                  <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-indigo-50 border border-indigo-100">
                    <Bot className="h-4.5 w-4.5 text-indigo-500" />
                  </div>
                  <div className="flex-1 min-w-0">
                    <div className="flex items-center justify-between">
                      <p className="font-bold text-slate-800 truncate">{agent.name}</p>
                      <span className="text-[10px] text-slate-400 font-medium shrink-0 ml-2">{agent.time}</span>
                    </div>
                    <p className="text-slate-500 text-[11px] truncate mt-0.5">{agent.detail}</p>
                  </div>
                  <Badge variant={agent.status === "completed" ? "success" : agent.status === "running" ? "info" : "destructive"} className="shrink-0 capitalize">
                    {agent.status}
                  </Badge>
                </div>
              ))}
            </div>
            <div className="mt-5 border-t border-slate-50 pt-4">
              <Link href="/agents" className="text-xs font-semibold text-[#1b59f8] hover:underline flex items-center gap-1">
                View All Agent Runs
                <ArrowUpRight className="h-3.5 w-3.5" />
              </Link>
            </div>
          </CardContent>
        </Card>

        {/* 2. Pending Approvals */}
        <Card className="border-slate-200">
          <CardHeader className="p-5 flex flex-row items-center justify-between border-b border-slate-100">
            <div>
              <CardTitle className="text-sm font-semibold">Pending Approvals</CardTitle>
              <CardDescription className="text-[10px]">Quality gate authorization requests</CardDescription>
            </div>
          </CardHeader>
          <CardContent className="p-4 pt-2">
            <div className="space-y-4">
              {pendingApprovals.map((item, i) => (
                <div key={item.title + i} className="flex items-start justify-between gap-3 text-xs py-1">
                  <div className="flex items-start gap-3 min-w-0">
                    <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-slate-50 border border-slate-200">
                      <FileText className="h-4.5 w-4.5 text-slate-500" />
                    </div>
                    <div className="min-w-0">
                      <p className="font-bold text-slate-800 truncate">{item.title}</p>
                      <p className="text-slate-500 text-[10px] mt-0.5 truncate">{item.subtitle}</p>
                    </div>
                  </div>
                  <div className="flex items-center gap-2 shrink-0">
                    <Badge variant="secondary" className="font-bold bg-slate-100 px-2 py-0.5">
                      {item.count}
                    </Badge>
                    <span className={cn("text-[9px] font-bold uppercase tracking-wider px-2 py-0.5 rounded border", item.priorityColor)}>
                      {item.priority.split(" ")[0]}
                    </span>
                  </div>
                </div>
              ))}
            </div>
            <div className="mt-6 border-t border-slate-50 pt-4">
              <Link href="/test-cases" className="text-xs font-semibold text-[#1b59f8] hover:underline flex items-center gap-1">
                View All Approvals
                <ArrowUpRight className="h-3.5 w-3.5" />
              </Link>
            </div>
          </CardContent>
        </Card>

        {/* 3. Recent Activity */}
        <Card className="border-slate-200">
          <CardHeader className="p-5 flex flex-row items-center justify-between border-b border-slate-100">
            <div>
              <CardTitle className="text-sm font-semibold">Recent Activity</CardTitle>
              <CardDescription className="text-[10px]">QA lifecycle audit logs</CardDescription>
            </div>
          </CardHeader>
          <CardContent className="p-4 pt-2">
            <div className="space-y-4">
              {recentActivities.map((act, i) => (
                <div key={i} className="flex items-start gap-3.5 text-xs py-1.5">
                  <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-slate-100 border border-slate-200 text-slate-600 font-bold text-xs">
                    {act.user.split(" ").map(n => n[0]).join("")}
                  </div>
                  <div className="flex-1 min-w-0">
                    <div className="flex items-center justify-between">
                      <p className="text-slate-800 font-medium">
                        <span className="font-bold">{act.user}</span> {act.action}
                      </p>
                      <span className="text-[10px] text-slate-400 font-medium shrink-0 ml-2">{act.time}</span>
                    </div>
                    <p className="text-slate-500 text-[10px] truncate mt-1">{act.subject}</p>
                  </div>
                </div>
              ))}
            </div>
            <div className="mt-5 border-t border-slate-50 pt-4">
              <Link href="/dashboard" className="text-xs font-semibold text-[#1b59f8] hover:underline flex items-center gap-1">
                View All Activity
                <ArrowUpRight className="h-3.5 w-3.5" />
              </Link>
            </div>
          </CardContent>
        </Card>
      </div>
    </div>
  );
}
